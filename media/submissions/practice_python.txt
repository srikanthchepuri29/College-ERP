POLYMORPHISM :-

class ironbox:
    def press(self):
        print(" pressing the ironbox")
class cotton_clothes(ironbox):
    def press(self):
        print("high tempereture for cotton clothes")
class silk_clothes(ironbox):
    def press(self):
        print("low tempereture for silk clothes")
        
b1 = cotton_clothes()
b2 = silk_clothes()
b1.press()
b2.press()
